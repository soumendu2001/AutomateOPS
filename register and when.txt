2.When: Displaying IP address of Ubuntu Servers:

---
- name: Demo of when
  hosts: all
  tasks:
    - name: Exploring FQDN when using conditional arguments
      ansible.builtin.command: hostname -I
      register: hostname_info
      when: ansible_distribution == 'Ubuntu'

    - name: Displaying FQDN of Ubuntu Servers
      ansible.builtin.debug:
        var: hostname_info.stdout
      when: ansible_distribution == 'Ubuntu' and hostname_info.changed
...

4) Install & (Re)Start Services Condition‑ally (Ubuntu vs Amazon Linux)

For Ubuntu Servers (Target2)
stop service
sudo systemctl status
sudo systemctl stop apache2

for Linux Servers:
stop httpd service
sudo systemctl status
sudo systemctl stop httpd

---
- name: Exploring when keyword
  hosts: target
  become: true
  gather_facts: true
  tasks:
    - name: Install Apache2 on Ubuntu Servers
      ansible.builtin.apt:
        name: apache2
        state: present
        update_cache: true
      when: ansible_distribution == 'Ubuntu'

    - name: Install httpd on Linux Server
      ansible.builtin.yum:
        name: httpd
        state: present
      when: ansible_distribution == 'Amazon'

    - name: Gather Service Facts
      ansible.builtin.service_facts:

    - name:  Restart httpd service if stopped (Linux)
      ansible.builtin.service:
        name: httpd
        state: restarted
        enabled: true
      when:  ansible_distribution == 'Amazon' and ansible_facts.services['httpd.service'] is defined and ansible_facts.services['httpd.service'].state in['stopped','inactive']
      notify: "httpd restarted"

    - name: Restart apache2  service if stopped (Ubuntu)
      ansible.builtin.service:
        name: apache2
        state: restarted
        enabled: true
      when: ansible_distribution == 'Ubuntu' and ansible_facts.services[ 'apache2.service'] is defined and ansible_facts.services[ 'apache2.service' ].state  in ['stopped' , 'inactive']
      notify: "apache2 restarted"

  handlers:
    - name: apache2 restarted
      listen: "apache2 restarted"
      ansible.builtin.debug:
           msg: "Apache2 service restarted"
    - name: httpd restarted
      listen: "httpd restarted"
      ansible.builtin.debug:
           msg: "httpd service restarted"
...

Create 2 python scripts:

show_service_disabled_and_inactive.sh 
=======================================================
#!/bin/sh
set -eu

# Print header
printf "%-60s %-12s %-12s\n" "SERVICE" "ENABLED" "ACTIVE"

# List services and evaluate enable/active state
systemctl list-unit-files --type=service --no-legend --no-pager \
| awk '{print $1}' \
| while read -r svc; do
  en="$(systemctl is-enabled "$svc" 2>/dev/null || echo "unknown")"
  ac="$(systemctl is-active  "$svc" 2>/dev/null || echo "unknown")"
  if [ "$en" = "disabled" ] && [ "$ac" = "inactive" ]; then
    printf "%-60s %-12s %-12s\n" "$svc" "$en" "$ac"
  fi
done | column -t

show_service_disabled_or_inactive.sh 
=====================================
#!/bin/sh
set -eu

printf "%-60s %-12s %-12s\n" "SERVICE" "ENABLED" "ACTIVE"

{
  # All disabled services from unit-files view
  systemctl list-unit-files --type=service --no-legend --no-pager \
  | awk '$2=="disabled"{print $1}'

  # All inactive services from units view
  systemctl list-units --type=service --state=inactive --no-legend --no-pager \
  | awk '{print $1}'
} | sort -u \
| while IFS= read -r svc; do
    en="$(systemctl is-enabled "$svc" 2>/dev/null || echo "unknown")"
    ac="$(systemctl is-active  "$svc" 2>/dev/null || echo "unknown")"
    printf "%-60s %-12s %-12s\n" "$svc" "$en" "$ac"
  done | column -t

6) Restart all stopped services

Let' see the services

---
- name: Demo to restart all stopped services
  hosts: target
  become: yes

  tasks:
    - name: Collect all Services
      ansible.builtin.service_facts:

    - name: Display all stopped services (safe across distros)
      ansible.builtin.debug:
        msg: "Name: {{ item.key }}, state: {{ item.value.state | default('unknown') }}, status: {{ item.value.status | default('unknown') }}"
      loop: "{{ ansible_facts.services | dict2items }}"
      loop_control:
        label: "{{ item.key }}"
      when:
        - (item.value.state | default( 'unknown')) in ['stopped', 'inactive']

    - name: Restart and enable all stopped or inactive services (with guards)
      ansible.builtin.service:
        name: "{{ item.key }}"
        state: restarted
        enabled: true
      loop: "{{ ansible_facts.services | dict2items }}"
      loop_control:
        label: "{{ item.key }}"
      when:
        - (item.value.state  | default( 'unknown')) in ['stopped', 'inactive']
        - (item.value.status | default( 'unknown')) not in [
            'static','not-found','indirect','inactive','alias','failed',
            'masked','generated','transient','enabled-runtime','active'
          ]
        - not (item.key is match( '@'))  # skip template/instance units
...'

7) Check if /tmp exists; create if missing
---
- name: Check if /tmp exists
  hosts: target
  become: yes
  gather_facts: false
  tasks:
    - name: Check if /tmp exists
      ansible.builtin.stat:
        path: /tmp
      register: tmp_exists

    - name: Confirming if /tmp exists
      ansible.builtin.debug:
        msg: |
         {% if tmp_exists.stat.exists %}
           /tmp exists
         {% else %}
           / tmp does not exists
         {% endif %}
    - name: Create /tmp if it does not exists 
      ansible.builtin.file:
        state: directory
        path: /tmp
      when: not tmp_exists.stat.exists
...
