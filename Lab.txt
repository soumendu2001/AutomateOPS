1. Declaring variable and displaying its value (Simple Variables)
# Every yaml file optionally can start with 3 dashes 
---
name: Declaring a variable and displaying
hosts: target
gather_facts: false
# vars: variables that will apply to the play, on all target systems
vars:
example_key_1: 'This is an example'
tasks:
 - name: Test Dictionary Key value
   ansible.builtin.debug:
            msg: "{{ example_key_1 }}"

# Every yaml file optionally can end with 3 dots

2. Declaring a dictionary and displaying its key and values (Dictionaries)

# Every yaml file optionally can start with 3 dashes
---
- name: Declaring a dictionary and displaying its key and values
  hosts: target
  gather_facts: false
# vars: variables that will apply to the play, on all target systems
  vars:
  dict:
   dict_key: 'This is a dictionary value'
  tasks:
  - name: Displaying Dictionary value
    ansible.builtin.debug:
      msg: "{{ dict }}"
  - name: Displaying dictionary key using dot notation
    ansible.builtin.debug:
      msg: "{{ dict.dict_key }}"
  - name: Displaying dictionary value using python bracket notation
    ansible.builtin.debug:
      msg: "{{ dict['dict_key'] }}"
	
# Every yaml file optionally can end with 3 dots
...

      - item1 
      - item2
      - item3
      - item4
      - item5
  tasks:
  - name: Displaying List value
    ansible.builtin.debug:
      msg: "{{ my_list }}"

  - name: Displaying 1st list item through dot notation
    ansible.builtin.debug:
      msg: "{{ my_list.0 }}"
  - name: Displaying 1st list item using python bracket notation
    ansible.builtin.debug:
       msg: "{{ my_list[0] }}"
  - name: Displaying 2nd list item using python dot notation

      msg: "{{ my_list.1 }}"

  - name: Displaying 2nd item using python bracket notation

  hosts: target
